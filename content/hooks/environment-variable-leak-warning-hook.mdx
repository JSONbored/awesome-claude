scriptBody: |-
  #!/usr/bin/env bash
  set -u
  if ! command -v jq >/dev/null 2>&1; then exit 0; fi
  input=$(cat)
  tool_name=$(printf '%s' "$input" | jq -r '.tool_name // .toolName // empty')
  case "$tool_name" in
    Write|Edit|MultiEdit|write|edit|multiedit) ;;
    *) exit 0 ;;
  esac
  file_path=$(printf '%s' "$input" | jq -r '.tool_input.file_path // .toolInput.file_path // .tool_input.path // .toolInput.path // empty')
  content=$(printf '%s' "$input" | jq -r '[.tool_input.content,.toolInput.content,.tool_input.new_string,.toolInput.new_string] | map(select(. != null and . != "")) | .[0] // empty')
  case "$file_path" in
    *.md|*.mdx|*.ts|*.tsx|*.js|*.jsx|*.py|*.go|*.yaml|*.yml|*.json) ;;
    *) exit 0 ;;
  esac
  [[ "$file_path" == *".env"* ]] && exit 0
  patterns=(
    'AKIA[0-9A-Z]{16}'
    'sk-[A-Za-z0-9]{20,}'
    'Bearer[[:space:]]+[A-Za-z0-9._-]{20,}'
    'ghp_[A-Za-z0-9]{20,}'
  )
  for pat in "${patterns[@]}"; do
    if printf '%s' "$content" | grep -Eiq "$pat"; then
      echo "Environment variable leak warning: possible secret-like token in $file_path" >&2
      exit 2
    fi
  done
  exit 0

## Overview

PreToolUse hook for Write, Edit, and MultiEdit that blocks edits when content
matches common live-secret patterns in source files.

## Troubleshooting

**Blocked placeholder in docs:** Use obvious placeholders like `YOUR_API_KEY` without live prefixes.
